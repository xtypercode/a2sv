class NumMatrix:

    def __init__(self, matrix: List[List[int]]):
        n, m = len(matrix), len(matrix[0])
        self.mres = [[0] * (m+1) for _ in range(n+1)]
        
        for i in range(n):
            for j in range(m):
                self.mres[i+1][j+1] = matrix[i][j] + self.mres[i+1][j] + self.mres[i][j+1] - self.mres[i][j]
        

    def sumRegion(self, row1: int, col1: int, row2: int, col2: int) -> int:
        return self.mres[row2+1][col2+1] - self.mres[row1][col2+1] - self.mres[row2+1][col1] + self.mres[row1][col1]
