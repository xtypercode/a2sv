class Solution:
    def maxVowels(self, s: str, k: int) -> int:
        count = ans = 0
        vowels = {"a", "e", "i", "o", "u"}

        i = 0
        for j in range(len(s)):
            while i < j and j - i + 1 > k:
                if s[i] in vowels:
                    count -= 1
                i += 1

            if s[j] in vowels:
                count += 1

            ans = max(count, ans)

        return ans
