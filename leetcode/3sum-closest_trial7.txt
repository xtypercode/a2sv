class Solution:
    def threeSumClosest(self, nums, target):
        nums.sort()
        ans = sum(nums[:3])

        for i in range(len(nums) - 2):
            left, right = i + 1, len(nums) - 1

            while left < right:
                s = nums[i] + nums[left] + nums[right]

                if abs(s - target) < abs(ans - target):
                    ans = s

                if s < target:
                    left += 1
                elif s > target:
                    right -= 1
                else:
                    return s

        return ans
